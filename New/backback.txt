  <script>
    let items = document.querySelectorAll('.slide-item');
    let textsLeft = document.querySelectorAll('.slide-text-left');
    let textsRight = document.querySelectorAll('.slide-text-right');
    let currentSlide = 0;
    let slideTimeout;
    document.querySelectorAll('video').forEach(v => v.load());
    function showSlide(index) {
      items.forEach(item => item.classList.remove('active'));
      textsLeft.forEach(text => text.classList.remove('active'));
      textsRight.forEach(text => text.classList.remove('active'));
      currentSlide = (index + items.length) % items.length;
      items[currentSlide].classList.add('active');
      textsLeft[currentSlide].classList.add('active');
      textsRight[currentSlide].classList.add('active');
      document.querySelectorAll('video').forEach(v => { v.pause(); v.currentTime = 0; });
      let currentItem = items[currentSlide];
      if (currentItem.getAttribute('data-type') === 'video') {
        let video = currentItem.querySelector('.main-media');
        video.play().catch(e => console.log("Video error: ", e));
        video.onended = () => { nextSlide(); };
      } else {
        clearTimeout(slideTimeout);
        slideTimeout = setTimeout(() => { nextSlide(); }, 8000); 
      }
    }
    function nextSlide() { showSlide(currentSlide + 1); }
    function prevSlide() { showSlide(currentSlide - 1); }
    document.getElementById('nextBtn').addEventListener('click', () => { clearTimeout(slideTimeout); nextSlide(); });
    document.getElementById('prevBtn').addEventListener('click', () => { clearTimeout(slideTimeout); prevSlide(); });
    if (items.length > 0) { showSlide(0); }
  </script>
